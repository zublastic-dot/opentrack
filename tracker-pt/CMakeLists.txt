include(opentrack-opencv)
find_package(OpenCV QUIET)
if(OpenCV_FOUND)
    foreach(k core calib3d features2d flann)
        otr_install_lib("opencv_${k}" "${opentrack-libexec}")
    endforeach()
    otr_module(tracker-pt-base STATIC)
    target_include_directories(${self} SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(${self} opentrack-cv opencv_core opencv_calib3d)
    if(MSVC)
        target_compile_options(${self} PRIVATE -EHsc -U_HAS_EXCEPTIONS -D_HAS_EXCEPTIONS=1)
    endif()
    set_property(TARGET ${self} PROPERTY OUTPUT_NAME "pt-base")
endif()
add_subdirectory(module)
